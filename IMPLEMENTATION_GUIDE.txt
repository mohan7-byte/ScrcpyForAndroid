    private void confirmScreenOffRemoteDevice() {
        getAttributes();
        if (TextUtils.isEmpty(serverAdr)) {
            Toast.makeText(context, "Server Address Empty", Toast.LENGTH_SHORT).show();
            return;
        }
        String[] serverInfo = Util.getServerHostAndPort(serverAdr);
        String device = serverInfo[0] + ":" + serverInfo[1];
        Dialog.displayDialog(this,
                getString(R.string.screen_off_title),
                getString(R.string.screen_off_ask, device),
                () -> screenOffRemoteDevice(device),
                () -> {
                });
    }

    private void screenOffRemoteDevice(String device) {
        Progress.showDialog(MainActivity.this, getString(R.string.screen_off_wait));
        ThreadUtils.workPost(() -> {
            AdbHelper.adbCmd(App.mContext, "connect", device);
            String result = AdbHelper.adbCmd(App.mContext, "-s", device, "shell", "input", "keyevent", "26");
            SessionLog.i("ADB screen off device=" + device + " result=" + result);
            ThreadUtils.post(() -> {
                Progress.closeDialog();
                if (MainActivity.this.isFinishing()) {
                    return;
                }
                Toast.makeText(context, getString(R.string.screen_off_sent, device), Toast.LENGTH_SHORT).show();
                startStatusMonitor();
            });
        });
    }
